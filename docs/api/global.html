| 'c:/dir/file.ext' | 'c:' | 'c:/dir' | '/dir'   | 'file.ext' | 'file' | '.ext' |
| '/dir/file.ext'   | ''   | '/dir'   | '/dir'   | 'file.ext' | 'file' | '.ext' |
